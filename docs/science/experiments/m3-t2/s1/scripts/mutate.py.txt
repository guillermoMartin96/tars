ALL_MUST_FAIL = {"M-22"}


def run(tag, src, nodes):
    xml = OUT / f"{tag}.xml"
    env = {"PYTHONPATH": str(src), "PYTHONDONTWRITEBYTECODE": "1", "PATH": "/usr/bin:/bin"}
    proc = subprocess.run([str(PROTO / ".venv/bin/python"), "-B", "-m", "pytest", "-q", "-p", "no:cacheprovider", f"--junitxml={xml}", *nodes],
                          cwd=PROTO, env=env, capture_output=True, text=True)
    (OUT / f"{tag}.stdout.txt").write_text(proc.stdout + proc.stderr)
    exits[tag] = proc.returncode
    res = []
    for tc in ET.parse(xml).getroot().iter("testcase"):
        f = tc.find("failure") if tc.find("failure") is not None else tc.find("error")
        res.append({"node": tc.get("name"), "status": "pass" if f is None else ("error" if tc.find("error") is not None else "fail"),
                    "message": None if f is None else (f.get("message") or "")[:200]})
    return res

for mid, edits, nodes, reason in M:
    tag = mid.split()[0]
    d = OUT / tag / "src"
    if d.exists():
        shutil.rmtree(d)
    shutil.copytree(PROTO / "src", d)
    for path, old, new in edits:
        p = d / path.removeprefix("src/")
        s = p.read_text()
        assert s.count(old) == 1, (mid, old)
        p.write_text(s.replace(old, new))
    res = run(tag, d, nodes)
    fails = [r for r in res if r["status"] != "pass"]
    if reason is None:
        verdict = "SURVIVED (expected equivalent)" if not fails else "KILLED (expected equivalent!)"
    elif not fails:
        verdict = "SURVIVED"
    elif tag in ALL_MUST_FAIL and len(fails) != len(res):
        verdict = "NOT_ALL_DESIGNATED_FAILED"
    else:
        verdict = "KILLED" if all(re.search(reason, f["message"] or "") for f in fails) else "KILLED_WRONG_REASON"
    report[mid] = {"nodes": nodes, "reason": reason, "verdict": verdict, "failed": len(fails), "total": len(res),
                   "pytest_exit": exits[tag], "edits": [{"path": p, "old": o, "new": n} for p, o, n in edits],
                   "results": res, "failures": fails}
    msgs = "; ".join(sorted({(f["message"] or "")[:60].replace("\n", " ") for f in fails}))
    print(f"{mid:62s} {verdict:30s} {len(fails):3d}/{len(res):3d}  {msgs[:200]}")
report["control_pytest_exit"] = exits["control"]
(OUT / "mutation_results.json").write_text(json.dumps(report, indent=1))
