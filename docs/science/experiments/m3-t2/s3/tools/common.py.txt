def now():
    return datetime.now(UTC).strftime("%Y-%m-%dT%H:%M:%SZ")


def sha(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


def fsha(p) -> str:
    return sha(Path(p).read_bytes())


def git_blob(b: bytes) -> str:
    return hashlib.sha1(b"blob %d\0" % len(b) + b).hexdigest()

class Runner:
    """Runs exact argv with an explicitly constructed env; writes <dir>/<NNN-id>.{json,stdout,stderr}."""

    def __init__(self, d: Path, start: int = 0):
        self.d = d
        self.n = start
        d.mkdir(parents=True, exist_ok=False)

    def run(self, sid, argv, cwd, env, timeout=3600):
        self.n += 1
        name = f"{self.n:03d}-{sid}"
        argv = [str(a) for a in argv]
        s = now()
        p = subprocess.run(argv, cwd=str(cwd), env=env, capture_output=True, timeout=timeout)
        e = now()
        (self.d / f"{name}.stdout").write_bytes(p.stdout)
        (self.d / f"{name}.stderr").write_bytes(p.stderr)
        rec = {"step": name, "argv": argv, "cwd": str(cwd), "env_passed": env, "utc_start": s,
               "utc_end": e, "exit": p.returncode, "stdout_sha256": sha(p.stdout),
               "stderr_sha256": sha(p.stderr), "label": LABEL}
        with open(self.d / f"{name}.json", "x") as fh:
            fh.write(json.dumps(rec, indent=1) + "\n")
        rec["stdout"], rec["stderr"] = p.stdout.decode(errors="replace"), p.stderr.decode(errors="replace")
        return rec


def export_bc(r: Runner, dest: Path, tar_path: Path, env: dict) -> dict:
    """git archive of exact BC into dest; verify every file against BC ls-tree blob ids."""
    genv = {**env, "GIT_CEILING_DIRECTORIES": ""}
    a = r.run("bc-archive", ["git", "-C", WT, "archive", "--format=tar", "-o", tar_path, BC], dest.parent, genv)
    ls = r.run("bc-ls-tree", ["git", "-C", WT, "ls-tree", "-r", "--full-tree", BC], dest.parent, genv)
    dest.mkdir()
    with tarfile.open(tar_path) as tf:
        tf.extractall(dest, filter="data")
    want = {ln.split("\t", 1)[1]: ln.split("\t", 1)[0].split()[2] for ln in ls["stdout"].splitlines()}
    got = {str(p.relative_to(dest)): git_blob(p.read_bytes()) for p in sorted(dest.rglob("*"))
           if p.is_file() or p.is_symlink()}
    mism = sorted(k for k in set(want) | set(got) if want.get(k) != got.get(k))
    data = (dest / "src/tars/sim/data/EarthExponentialAtmosphereData.txt").read_bytes()
    return {"ok": a["exit"] == 0 and ls["exit"] == 0 and not mism and sha(data) == PIN
            and git_blob(data) == DATA_BLOB and fsha(dest / "uv.lock") == LOCK_PIN,
            "tar_sha256": fsha(tar_path), "files": len(got), "blob_mismatches": mism,
            "data_sha256": sha(data), "uv_lock_sha256": fsha(dest / "uv.lock"),
            "manifest": got}


def check_env_record(path, expect_prefix=None, expect_subject=None) -> tuple[bool, dict]:
    """Completeness of one own-process ENV record (the declared contract)."""
    try:
        e = json.loads(Path(path).read_text())
    except (OSError, json.JSONDecodeError) as exc:
        return False, {"error": f"missing/unreadable: {exc}"}
    s2 = e.get("s2_env_record", {})
    b = s2.get("body") or {}
    run = b.get("run", {})
    cpu = b.get("cpu", {})
    libm = b.get("libm behaviour digest", {})
    py = b.get("python", {})
    body_text = json.dumps(b, sort_keys=True, default=str) if b else ""
    f = {
        "pid": e.get("pid"), "observer": e.get("observer"), "label": e.get("label"),
        "s2_error": s2.get("error"),
        "s2_body_sha256_verifies": bool(b) and sha(body_text.encode()) == s2.get("S2-ENV-RECORD body sha256"),
        "libm_arguments": libm.get("[O] arguments"), "libm_results": libm.get("[O] results hashed"),
        "libm_sha256": libm.get("[O] sha256"),
        "cpu_brand": cpu.get("[O] brand"),
        "cpu_artificial_U": sorted(k for k, v in cpu.items() if isinstance(v, str) and "No such file" in v),
        "fp_fields": len(b.get("floating_point", {})),
        "build_config_present": bool(b.get("build_config [P]")),
        "exe_realpath_in_body": py.get("[O] executable realpath"),
        "exe_sha_in_body": py.get("[O] executable sha256"),
        "libpython_sha_in_body": py.get("[P] libpython sha256"),
        "lock_in_body": run.get("[O] uv.lock sha256"),
        "lock_independent": e.get("subject_uv_lock_sha256_independent"),
        "commit_field": run.get("[O] commit (git rev-parse HEAD)"),
        "executable_realpath": e.get("executable_realpath"), "executable_sha256": e.get("executable_sha256"),
        "libpython_sha256": e.get("libpython_sha256"), "sys.version": e.get("sys.version"),
        "sys.prefix": e.get("sys.prefix"), "subject_root": e.get("subject_root"),
        "label_p5": e.get("label_p5"),
    }
    commit = f["commit_field"] or ""
    ok = (not f["s2_error"] and f["s2_body_sha256_verifies"] and f["libm_arguments"] == 6235
          and f["libm_results"] == 7175 and isinstance(f["cpu_brand"], str) and f["cpu_brand"].startswith("[O] ")
          and not f["cpu_artificial_U"] and f["fp_fields"] > 0 and f["build_config_present"]
          and f["exe_realpath_in_body"] == PY and f["exe_sha_in_body"] == "[O] " + PY_SHA
          and f["libpython_sha_in_body"] == "[O] " + LIBPY_SHA
          and f["lock_in_body"] == "[O] " + LOCK_PIN and f["lock_independent"] == LOCK_PIN
          and f["executable_realpath"] == PY and f["executable_sha256"] == PY_SHA
          and f["libpython_sha256"] == LIBPY_SHA and str(f["sys.version"]).startswith("3.12.15 ")
          and commit.startswith("[U] unavailable") and "not a git repository" in commit
          and f["label_p5"] == LABEL
          and (expect_subject is None or Path(f["subject_root"]) == Path(expect_subject))
          and (expect_prefix is None or Path(f["sys.prefix"]) == Path(expect_prefix)))
    f["ok"] = bool(ok)
    return bool(ok), f
