class Stage:
    def __init__(self, name, start):
        self.name, self.res, self.ok = name, {"record": f"S3 corrected packaging {name}", "started": C.now(),
                                              "label": C.LABEL, "checks": {},
                                              "subject": {"bc_sha": C.BC, "bc_tree": C.BC_TREE}}, True
        self.r = C.Runner(OUT / f"steps-{name}", start) if not (OUT / f"steps-{name}").exists() else None
        if self.r is None:
            raise SystemExit(f"occupied: {OUT / f'steps-{name}'}")

    def check(self, name, cond, **facts):
        self.ok &= bool(cond)
        self.res["checks"].setdefault(name, []).append({"pass": bool(cond), **facts})

    def finish(self):
        if not C.os.environ.get("GIT_CEILING_DIRECTORIES"):
            raise SystemExit("STOP: driver process lacks GIT_CEILING_DIRECTORIES (its ENV commit field would "
                             "report the enclosing evidence repository)")
        # The driver process's own ENV record, captured after its payload (Q-1: artifact-check driver).
        envp = OUT / f"driver-{self.name}.env.json"
        spec = importlib.util.spec_from_file_location("_envcap2", str(C.ENVCAP))
        m = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(m)
        m.capture(str(envp), str(TREE), f"packaging driver {self.name}", "driver-finally")
        good, facts = C.check_env_record(envp, expect_subject=TREE)
        self.check("DRIVER-ENV", good and facts["sys.prefix"] == str(Path(C.PY).parents[1]), **facts)
        self.res["finished"] = C.now()
        self.res["ok"] = bool(self.ok)
        with open(OUT / f"{self.name.upper()}.json", "x") as fh:
            fh.write(json.dumps(self.res, indent=1, default=str) + "\n")
        print(json.dumps({"ok": self.res["ok"], "stop": self.res.get("stop"),
                          "checks": {k: [c["pass"] for c in v] for k, v in self.res["checks"].items()}}))
        sys.exit(0 if self.ok else 1)


def need(prev, base=None):
    p = (base or OUT) / f"{prev.upper()}.json"
    d = json.loads(p.read_text())
    if not d["ok"]:
        raise SystemExit(f"STOP: {prev} did not pass")
    return d, C.fsha(p)

def hook_names(stderr):
    return sorted(re.search(r"Calling `uv_build\.(\w+)\(", ln).group(1)
                  for ln in stderr.splitlines() if "Calling `uv_build." in ln)


def hook_fn(argv):
    code = (argv or [""])[-1]
    m = re.search(r"backend\.(build_\w+)\(", code) or re.search(r"getattr\(backend, \"(\w+)\"", code)
    return m.group(1) if m else None

                 member_sha256=C.sha(data))
    (OUT / "LISTINGS.json").write_text(json.dumps(listings, indent=1) + "\n")

    def sp(venv):
        return venv / "lib/python3.12/site-packages"

    def wrap(payload):
        return WRAP.format(payload=payload, envcap=str(C.ENVCAP))

    def make_venv(tag, inst, whl, cache):
        venv = SCR / f"venv-{tag}"
        tool, envs = [], []
        if inst == "uv":
            tool.append(st.r.run(f"{tag}-uv-venv", [C.UV, "venv", "-v", "--offline", "--no-python-downloads",
                                                    "--python", C.PY, "--cache-dir", cache, venv], SCR, env))
            tool.append(st.r.run(f"{tag}-uv-pip-install", [C.UV, "pip", "install", "-v", "--offline", "--no-deps",
                                                           "--no-index", "--python", venv / "bin/python",
                                                           "--cache-dir", cache, whl], SCR, env))
            cfg = (venv / "pyvenv.cfg").read_text() if (venv / "pyvenv.cfg").exists() else None
            home = re.search(r"^home = (.+)$", cfg or "", re.M)
            # uv writes the minor-version link dir (cpython-3.12-…/bin); bind by realpath (run-2 checker
            # compared the literal string and failed although the venvs resolve to 3.12.15).
            bound = bool(home) and C.os.path.realpath(home.group(1) + "/python3.12") == C.PY
            return venv, tool, envs, {"pyvenv.cfg": cfg, "interpreter_bound": bound}
        p, g, f = pyrun(f"{tag}-venv", [C.PY, "-I", "-c", wrap(RUNMOD.format(mod="venv")), venv], SCR,
                        f"{tag} python -m venv (setup)")
        envs.append((p, g, f))
        p, g, f = pyrun(f"{tag}-pip-install", [venv / "bin/python", "-I", "-c", wrap(RUNMOD.format(mod="pip")),
                                               "install", "--no-deps", "--no-index", whl], SCR, f"{tag} pip install", venv)
        envs.append((p, g, f))
        p, g, f = pyrun(f"{tag}-pip-version", [venv / "bin/python", "-I", "-c", wrap(RUNMOD.format(mod="pip")),
                                               "--version"], SCR, f"{tag} pip --version", venv)
        envs.append((p, g, f))
        cfg = (venv / "pyvenv.cfg").read_text()
        return venv, tool, envs, {"pyvenv.cfg": cfg, "interpreter_bound": True,
                                  "ensurepip_children": "incidental helper processes of `python -m venv` "
                                                        "(ensurepip); inventoried, no ENV (Q-1)"}

    forbidden = [C.WT.resolve(), TREE.resolve(), Path("/Users/guillermomartin/projects/tars").resolve()]
    for name, whl in wheels.items():
        for inst in ("uv", "pip"):
            tag = f"{name}-{inst}"
            venv, tool, envs, vf = make_venv(tag, inst, whl, SCR / f"cache-{tag}")
            rec = sp(venv) / "tars-0.1.0.dist-info/RECORD"
            inst_txt = sp(venv) / "tars-0.1.0.dist-info/INSTALLER"
            st.check("PK-4", all(t["exit"] == 0 for t in tool) and all(p["exit"] == 0 and g for p, g, _ in envs)
                     and rec.exists() and vf["interpreter_bound"], wheel=name, installer=inst, venv=str(venv),
                     tool_steps=[t["step"] for t in tool], process_env=[f for _, _, f in envs],
                     installer_file=inst_txt.read_text().strip() if inst_txt.exists() else None,
                     installed_record=rec.read_text() if rec.exists() else None,
                     installer_version=(envs[2][0]["stdout"].strip() if inst == "pip"
                                        else "uv frontend (tool); version/sha256 in stage0 steps"), **vf)
            cwd = SCR / f"cwd-{tag}"
            cwd.mkdir()
            ri, g, facts = pyrun(f"PK-5-{tag}-import", [venv / "bin/python", "-I", "-c", wrap(IMPORT)], cwd,
                                 f"PK-5 {tag} import", venv)
            o = last_json(ri["stdout"])
            spr = sp(venv).resolve()
            paths = [o.get("tars_file"), o.get("component_file"), o.get("data_path")]
            installed = sp(venv) / MEMBER
            st.check("PK-5", ri["exit"] == 0 and all(x and Path(x).is_relative_to(spr) for x in paths)
                     and not any(x and Path(x).is_relative_to(f) for x in paths for f in forbidden)
                     and o.get("data_sha256") == C.PIN and installed.exists() and C.fsha(installed) == C.PIN
                     and o.get("density_0") == o.get("row0_rho0") and o.get("density_0")
                     and o.get("executable_realpath") == C.PY and g,
                     wheel=name, installer=inst, step=ri["step"], observed=o, process_env=facts)
    for variant, expected in (("delete", "missing"), ("flip", "digest_mismatch"), ("crlf", "digest_mismatch")):
        tag = f"pk6-{variant}"
        venv, tool, envs, vf = make_venv(tag, "uv", wheels["tree"], SCR / f"cache-{tag}")
        data_p = sp(venv) / MEMBER
        raw = data_p.read_bytes()
        if variant == "delete":
            data_p.unlink()
            edit = {"deleted": True}
        elif variant == "flip":
            n = raw.count(b"60.0, 3.206e-4,")
            data_p.write_bytes(raw.replace(b"60.0, 3.206e-4,", b"60.0, 3.207e-4,", 1))
            edit = {"occurrences": n, "after_sha256": C.fsha(data_p)}
        else:
            data_p.write_bytes(raw.replace(b"\n", b"\r\n"))
            edit = {"after_sha256": C.fsha(data_p), "after_size": data_p.stat().st_size}
        cwd = SCR / f"cwd-{tag}"
        cwd.mkdir()
        rp, g, facts = pyrun(f"PK-6-{variant}", [venv / "bin/python", "-I", "-c", wrap(PROFILED)], cwd,
                             f"PK-6 {variant}", venv)
        o = last_json(rp["stdout"])
        st.check("PK-6", all(t["exit"] == 0 for t in tool) and vf["interpreter_bound"] and o.get("outcome") == "raised"
                 and o.get("reason") == expected and o.get("calls") == 0 and g
                 and (variant != "flip" or edit["occurrences"] == 1),
                 variant=variant, expected=expected, edit=edit, step=rp["step"], observed=o, process_env=facts)
    cwd = SCR / "cwd-pk7"
    cwd.mkdir()
    rz, g, facts = pyrun("PK-7-zipimport", [C.PY, "-I", "-c", wrap(ZIPIMPORT), wheels["tree"]], cwd, "PK-7 zipimport")
    o = last_json(rz["stdout"])
    st.check("PK-7", o.get("reason") == "unreadable" and g and facts["sys.prefix"] == str(Path(C.PY).parents[1]),
             recorded_outcome=o, step=rz["step"], process_env=facts,
             expected="reason unreadable (fail closed); zip import stays unsupported")
    for name in wheels:                         # PK-8 (in this driver process)
        src = (sp(SCR / f"venv-{name}-uv") / "tars/sim/atmosphere.py").read_text()
        mods = set()
        for node in ast.walk(ast.parse(src)):
            if isinstance(node, ast.Import):
                mods |= {a.name.split(".")[0] for a in node.names}
            elif isinstance(node, ast.ImportFrom):
                mods.add((node.module or "").split(".")[0])
        st.check("PK-8", mods <= set(sys.stdlib_module_names), wheel=name, imports=sorted(mods))
    s = st.r.run("wt-status-end", ["git", "-C", C.WT, "status", "--porcelain=v1", "--untracked-files=all"], SCR,
                 {**env, "GIT_CEILING_DIRECTORIES": ""})
    st.check("SRC-worktree-end", s["exit"] == 0 and s["stdout"] == "", status=s["stdout"])
    st.check("TOOL-python-end", C.fsha(C.PY) == C.PY_SHA)
    st.finish()
