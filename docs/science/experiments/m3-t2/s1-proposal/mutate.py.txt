def run(tag, src, nodes):
    xml = OUT / f"{tag}.xml"
    env = {"PYTHONPATH": str(src), "PYTHONDONTWRITEBYTECODE": "1", "PATH": "/usr/bin:/bin"}
    subprocess.run([str(PROTO / ".venv/bin/python"), "-B", "-m", "pytest", "-q", "-p", "no:cacheprovider", f"--junitxml={xml}", *nodes], cwd=PROTO, env=env, capture_output=True, text=True)
    res = []
    for tc in ET.parse(xml).getroot().iter("testcase"):
        f = tc.find("failure") if tc.find("failure") is not None else tc.find("error")
        res.append({"node": tc.get("name"), "status": "pass" if f is None else tc.find("error") is not None and "error" or "fail", "message": None if f is None else (f.get("message") or "")[:160]})
    return res

assert all(r["status"] == "pass" for r in report["control"]), report["control"]

    tag = mid.split()[0]
    d = OUT / tag / "src"
    if d.exists(): shutil.rmtree(d)
    shutil.copytree(PROTO / "src", d)
    p = d.parent / path; p = d / path.removeprefix("src/")
    s = p.read_text()
    for o, n in [(old, new), *extra]:
        assert s.count(o) == 1, (mid, o); s = s.replace(o, n)
    p.write_text(s)
    res = run(tag, d, nodes)
    fails = [r for r in res if r["status"] != "pass"]
    report[mid] = {"nodes": nodes, "failed": len(fails), "total": len(res), "failures": fails}
    print(f"{mid:55s} failed {len(fails):3d}/{len(res):3d}  " + ("; ".join(sorted({(f['message'] or '')[:70] for f in fails}))[:220]))
