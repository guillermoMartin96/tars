"""I-105 census validator (v11 §3.10 census record; I104-01).

Usage: python -I validate_census.py <listing.txt> <census.csv> <kind: ci|host> <out.json> [<schema-1 csv> <label>]...

Checks, for a schema-2 census: one record per listing entry, in listing order; set, index, the
x/y/z hex input and the listing label equal the frozen listing's own text exactly; source is
"listing label" for a geometric reject and "twin record (G-08 adapter)" for a passes entry; a
reject's outcome is its label and its iteration/final-step (and, for ci, h/phi) fields are
not_evaluated; a passes entry has an integer iteration count and a hex final step.
Optional comparison: each schema-1 census given is compared with this census PROJECTED onto the
schema-1 columns (the hex input dropped). This is a projection comparison computed here; no
enriched schema-1 record is written, and no historical file is read for anything else.
"""
import csv, hashlib, json, sys
from collections import Counter

listing, census, kind, out = sys.argv[1:5]
pairs = sys.argv[5:]
L = open(listing).read().split("\n")
hdr, L = L[0], [ln.split(",") for ln in L[1:-1]]
rows = list(csv.reader(open(census, newline="")))
head, body = rows[0], rows[1:]
COLS = {"ci": "set,index,x_hex,y_hex,z_hex,label,outcome,iterations,final_step_hex,h_hex,phi_hex,source".split(","),
        "host": "set,index,x_hex,y_hex,z_hex,label,outcome,iterations,final_step_hex,source,component_outcome".split(",")}[kind]
res = {"census": census, "census_sha256": hashlib.sha256(open(census, "rb").read()).hexdigest(), "kind": kind,
       "listing_sha256": hashlib.sha256(open(listing, "rb").read()).hexdigest(), "header_ok": head == COLS,
       "records": len(body), "listing_entries": len(L)}
I = {c: i for i, c in enumerate(COLS)}
problems = Counter()
outc = Counter()
for n, (r, e) in enumerate(zip(body, L)):
    if len(r) != len(COLS):
        problems["field count"] += 1
        continue
    if [r[I["set"]], r[I["index"]], r[I["x_hex"]], r[I["y_hex"]], r[I["z_hex"]], r[I["label"]]] != e:
        problems["set/index/hex input/label differs from listing"] += 1
    outc[r[I["outcome"]]] += 1
    if e[5] != "passes":
        ne = [r[I["iterations"]], r[I["final_step_hex"]]] + ([r[I["h_hex"]], r[I["phi_hex"]]] if kind == "ci" else [])
        if r[I["outcome"]] != e[5]: problems["reject outcome != label"] += 1
        if any(v != "not_evaluated" for v in ne): problems["reject with evaluated field"] += 1
        if r[I["source"]] != "listing label": problems["reject source"] += 1
        if kind == "host" and r[I["component_outcome"]] != e[5]: problems["host component outcome != label"] += 1
    else:
        if r[I["source"]] != "twin record (G-08 adapter)": problems["passes source"] += 1
        if not r[I["iterations"]].isdigit(): problems["passes iterations not integer"] += 1
        try: float.fromhex(r[I["final_step_hex"]])
        except ValueError: problems["passes final step not hex"] += 1
        if r[I["outcome"]] in ("origin", "evolute_region"): problems["passes entry given a geometric outcome"] += 1
        if kind == "ci":
            acc = r[I["outcome"]] == "accepted"
            for c in ("h_hex", "phi_hex"):
                v = r[I[c]]
                if acc:
                    try: float.fromhex(v)
                    except ValueError: problems[f"accepted {c} not hex"] += 1
                elif v != "discarded": problems[f"non-accepted {c} not discarded"] += 1
        if kind == "host" and r[I["component_outcome"]] != r[I["outcome"]]: problems["host component != census"] += 1
res["record_count_equals_listing"] = len(body) == len(L)
res["outcomes"] = dict(sorted(outc.items()))
res["labels"] = dict(sorted(Counter(e[5] for e in L).items()))
res["problems"] = dict(problems)
res["valid"] = res["header_ok"] and res["record_count_equals_listing"] and not problems
comps = []
for old, label in zip(pairs[0::2], pairs[1::2]):
    o = list(csv.reader(open(old, newline="")))
    oh, ob = o[0], o[1:]
    keep = [c for c in oh if c in I or c == "census_outcome"]
    proj = [[r[I["outcome" if c == "census_outcome" else c]] for c in oh] for r in body]
    diff = sum(a != b for a, b in zip(ob, proj)) + abs(len(ob) - len(proj))
    comps.append({"schema-1 census": old, "label": label, "sha256": hashlib.sha256(open(old, "rb").read()).hexdigest(),
                  "schema-1 columns": oh, "rows": len(ob), "rows differing from the schema-2 projection": diff,
                  "method": "schema-2 rows projected onto the schema-1 columns (hex input dropped); comparison only, no join written"})
res["projection_comparisons"] = comps
json.dump(res, open(out, "w"), indent=1)
print(json.dumps({k: res[k] for k in ("records", "header_ok", "outcomes", "problems", "valid")}, indent=1),
      json.dumps([{k: c[k] for k in ("label", "rows", "rows differing from the schema-2 projection")} for c in comps]))
sys.exit(0 if res["valid"] and all(c["rows differing from the schema-2 projection"] == 0 for c in comps) else 1)
