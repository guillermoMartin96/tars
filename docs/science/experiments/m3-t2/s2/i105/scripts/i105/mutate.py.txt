"""S2 commit D: the §3.12 mutation campaign (proposal v11; I-40 §10 and S1 precedent).

Usage: mutate.py TREE_ROOT OUT_DIR, where TREE_ROOT is a fresh clone at commit BC with its
.venv synced. Mutants are applied to copies of `src/` under OUT_DIR, never to the clone or
the implementation worktree, and run through PYTHONPATH.

A mutant is KILLED only if at least one designated node fails and every failure is
assertion-type (AssertionError, a bare assert, or "Failed: DID NOT RAISE"). A failure by any
other exception is KILLED_WRONG_REASON (not a kill). A mutant with no failing designated node
is SURVIVED, which is stop 8 unless it is one of the documented equivalent survivors (each
with its written proof in the derivation note). The unmutated control must pass every
designated node. JUnit is recorded for every run.

I-105 (I104-04, I104-02):
- MG-02 and MG-04 are bound to their exact v11 §3.12 witnesses, each alone in its own node:
  G-04 (0, nan, 7e6) and the G-03/G-07 accepted axis point (0, 0, 7.0e6). The transformations are
  unchanged. A kill of either also requires the exact witness in the assertion message (WITNESS).
- Every execution (control and each mutant) also runs the nine component-free recorders (ENV,
  X-P3/RP-9 listing, G-17a, G-17b) in the same pytest process, and gets its own run record
  (run_record.py with the same interpreter) and an extraction status. A recorder node that does
  not pass, or a record that is missing or fails its digest, makes that execution INCOMPLETE
  (stop 18), which is NOT OK. Recorder nodes are never designated nodes.
- Failure messages are kept in full (no 300-character truncation).
"""

    return "".join("    " + ln if ln.strip() else ln for ln in block.splitlines(keepends=True))

def run(tag, src, nodes):
    xml = OUT / f"{tag}.xml"
    env = {"PYTHONPATH": str(src), "PYTHONDONTWRITEBYTECODE": "1", "PATH": "/usr/bin:/bin"}
    py = str(PROTO / ".venv/bin/python")
    rr = subprocess.run([py, "-I", str(SCRIPTS / "run_record.py"), str(PROTO), str(OUT / f"{tag}.run-record.json"),
                         f"I-105 campaign execution {tag}; component source {src}", *PROBE_PATHS],
                        capture_output=True, text=True)
    rec = json.loads((OUT / f"{tag}.run-record.json").read_text())
    rec["executed component"] = {"path": str(Path(src) / "tars/sim/geodesy.py"),
                                 "sha256": hashlib.sha256((Path(src) / "tars/sim/geodesy.py").read_bytes()).hexdigest()}
    cmd = [py, "-B", "-m", "pytest", "-q", "-p", "no:cacheprovider", f"--junitxml={xml}", *nodes, *RECORDERS]
    rec["command"] = cmd
    rec["start utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    proc = subprocess.run(cmd, cwd=PROTO, env=env, capture_output=True, text=True)
    rec["end utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    log = OUT / f"{tag}.stdout.txt"
    log.write_text(proc.stdout + proc.stderr)
    ex = subprocess.run([py, "-I", str(SCRIPTS / "extract_s2_records_v2.py"), str(log), str(OUT / f"{tag}.records")],
                        capture_output=True, text=True)
    status = json.loads((OUT / f"{tag}.records" / "extraction-status.json").read_text())
    rec.update({"pytest exit": proc.returncode, "probe exit": rr.returncode,
                "stdout sha256": hashlib.sha256(log.read_bytes()).hexdigest(),
                "junit sha256": hashlib.sha256(xml.read_bytes()).hexdigest(),
                "records complete": status.get("complete") is True, "extract exit": ex.returncode})
    (OUT / f"{tag}.run-record.json").write_text(json.dumps(rec, indent=1) + "\n")
    exits[tag] = proc.returncode
    res, recs = [], []
    for tc in ET.parse(xml).getroot().iter("testcase"):
        f = tc.find("failure") if tc.find("failure") is not None else tc.find("error")
        row = {"node": f"{tc.get('classname')}::{tc.get('name')}",
               "status": "pass" if f is None else ("error" if tc.find("error") is not None else "fail"),
               "message": None if f is None else (f.get("message") or "")}
        (recs if tc.get("classname") in RECORDER_CLASSES else res).append(row)
    complete = (rec["records complete"] and rr.returncode == 0 and len(recs) == 9
                and all(r["status"] == "pass" for r in recs))
    records[tag] = {"complete": complete, "recorder_nodes": len(recs),
                    "recorder_nodes_passed": sum(r["status"] == "pass" for r in recs),
                    "records": {k: status.get(k) for k in ("S2-ENV-RECORD", "S2-G17A-REPORT", "S2-G17B-RECORD")},
                    "listing": status.get("S2-XP3-LISTING"), "run_record": f"{tag}.run-record.json"}
    return res
