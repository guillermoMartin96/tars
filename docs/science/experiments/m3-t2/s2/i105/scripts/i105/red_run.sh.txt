#!/bin/bash
# I-105 supplemental red run (I104-04 test change; I104-01 emitter change). Fresh clone of commit
# A from GitHub; src/tars/sim/geodesy.py absent; constants.py unchanged (K); the two frozen test
# files installed. One pytest execution runs the component test file together with the nine
# component-free recorders (ENV, X-P3/RP-9 listing, G-17a, G-17b), so this red run carries its
# own DL-5 record, listing and run record (v11 §3.11 ENV, §3.16).
set -u
E=$HOME/tars-evidence/I-105; O=$E/red; C=$E/clone-a-red; S=$E/scripts
A=6c05e53c5ffcb892dacaa6f84e12a163830a4119
export PYTHONDONTWRITEBYTECODE=1 PATH="$HOME/Library/Python/3.9/bin:/usr/bin:/bin:/usr/sbin:/sbin" UV_PYTHON_DOWNLOADS=never
mkdir -p "$O"; date -u +%FT%TZ > "$O/start-utc.txt"
git clone -q https://github.com/guillermoMartin96/tars.git "$C" && git -C "$C" checkout -q --detach "$A" || exit 98
cd "$C" || exit 99
uv sync --locked -q > "$O/uv-sync.log" 2>&1; echo $? > "$O/uv-sync.exit"
cp "$E/frozen/tests/test_s2_geodetic.py" tests/test_s2_geodetic.py
cp "$E/frozen/tests/test_s2_xp3_listing.py" tests/test_s2_xp3_listing.py
git rev-parse HEAD > "$O/head.txt"
git status --porcelain --untracked-files=all > "$O/git-status-porcelain.txt"
for f in tests/test_s2_geodetic.py tests/test_s2_xp3_listing.py; do
  echo "$(git hash-object $f) $(shasum -a 256 $f | cut -d' ' -f1) $f"; done > "$O/test-file-identities.txt"
{ echo "src/tars/sim/geodesy.py present: $(test -e src/tars/sim/geodesy.py && echo YES || echo no)"
  echo "git ls-files src/tars/sim/geodesy.py: '$(git ls-files src/tars/sim/geodesy.py)'"
  echo "constants.py blob $(git hash-object src/tars/sim/constants.py) (K bcda30e2d008a008986b58790820d7b3de12fd68)"; } > "$O/absence-proof.txt"
.venv/bin/python -I "$S/run_record.py" "$C" "$O/run-record.json" "I-105 supplemental red run (A + 2 frozen test files)" \
  tests/test_s2_geodetic.py tests/test_s2_xp3_listing.py tests/test_s2_environment_record.py tests/test_s2_dl4_points.py \
  src/tars/sim/geodesy.py src/tars/sim/constants.py tests/s2_dl4_points.txt > "$O/run-record.stdout" 2>&1
.venv/bin/python -B -m pytest -v -rA -p no:cacheprovider --junitxml="$O/red-junit.xml" \
  tests/test_s2_geodetic.py tests/test_s2_environment_record.py tests/test_s2_xp3_listing.py tests/test_s2_dl4_points.py \
  > "$O/red-pytest.log" 2>&1
echo $? > "$O/red-pytest.exit"
.venv/bin/python -I "$S/classify_red_v2.py" "$O/red-junit.xml" "$HOME/tars-evidence/I-102/supp-red/red-classification.json" \
  "$O/red-classification.json" > "$O/classify.stdout" 2>&1; echo $? > "$O/classify.exit"
.venv/bin/python -I "$S/extract_s2_records_v2.py" "$O/red-pytest.log" "$O/records" > "$O/extract.stdout" 2>&1; echo $? > "$O/extract.exit"
date -u +%FT%TZ > "$O/end-utc.txt"; touch "$O/DONE"
