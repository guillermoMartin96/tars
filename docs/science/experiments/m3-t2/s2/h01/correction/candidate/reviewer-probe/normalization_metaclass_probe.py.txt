from tars.sim.geodesy import geodetic
calls=[]
sentinel=ValueError('unsupported component metaclass __name__ sentinel')
class Meta(type):
    def __getattribute__(cls,name):
        if name=='__name__':
            calls.append(name)
            raise sentinel
        return super().__getattribute__(name)

    pass

assert outcome['kind']=='raised' and outcome['is_sentinel'] and not outcome['is_TypeError'] and calls==['__name__']
