import sys,subprocess,shutil,json
from pathlib import Path

R=Path.cwd();S=R/'review_scratch';D=R/'docs/science/experiments/m3-t0';sys.path.insert(0,str(D))
import mutation_check_review02 as h
names=['json_reviewer_R2F3_mutant','r4_absolute_part_removed','r5_conversion_bound_relative_only','r6_zero_b_case_removed','r6f2_up_overflow_restored']
def run(name):
 root=S/'mutants'/name;shutil.copytree(R/'docs',root/'docs',ignore=shutil.ignore_patterns('__pycache__'),dirs_exist_ok=True);d=root/h.REL
 corr,expected=h.MUTANTS[name];corr(d)
