import os,subprocess,pathlib,concurrent.futures
R=pathlib.Path.cwd(); O=R/'review_scratch/reviewer'; P=R/'.venv/bin/python'
def run(name,cmd,cwd,src):
 env=dict(os.environ,PYTHONPATH=str(src),PYTHONDONTWRITEBYTECODE='1')
 with (O/(name+'.txt')).open('w') as f:
  r=subprocess.run([str(P),'-B',*cmd],cwd=cwd,env=env,stdout=f,stderr=subprocess.STDOUT)
 return name,r.returncode
jobs=[]
for tree in ['base','v2','v3s','v3l']:
 t=O/tree
 jobs.append((tree+'-s1',['-m','pytest','-q','-p','no:cacheprovider','tests/test_s1_failure_paths.py','tests/test_s1_force_context.py'],t,t/'src'))
 jobs.append((tree+'-integrators',[str(R/'review_scratch/pkg/harness_integrators.py'),str(O/(tree+'.jsonl'))],t,t/'src'))
jobs.append(('v3s-full',['-m','pytest','-q','-p','no:cacheprovider'],O/'v3s',O/'v3s/src'))
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as ex:
 for r in ex.map(lambda x:run(*x),jobs):print(r,flush=True)
