"""I-33 F2 evidence (research, NOT T2 evidence): premise-checker controls derived from
A-FP-2's exact inequality |fhat - f| <= UF |f| + nu, UF = 2 x 2^-52, nu = 2 x 2^-1074
(t0_termination.py lines 455-457, 474-475).

Every expected outcome below is computed from the closed-form window
(pog_lib.premise_window), which is derived separately from the decision procedure under
test (pog_lib.premise, convexity at the enclosure endpoints).
Usage: python -I f2_premise_controls.py
"""
import math
import sys
from fractions import Fraction as Q
from pathlib import Path

sys.dont_write_bytecode = True
sys.path.insert(0, str(Path(__file__).resolve().parent))
import pog_lib as L  # noqa: E402


def ulps_from(f: float, k: int) -> float:
    for _ in range(abs(k)):
        f = math.nextafter(f, math.inf if k > 0 else -math.inf)
    return f


def last_inside_and_first_outside(lo: Q, hi: Q) -> tuple[float, float, float, float]:
    """The doubles just inside / just outside the PASS window, from the closed form."""
    plo, phi, fbelow, fabove = L.premise_window(lo, hi)
    up_in = float(phi)
    if Q(up_in) > phi:
        up_in = math.nextafter(up_in, -math.inf)
    up_out = math.nextafter(up_in, math.inf)
    dn_in = float(plo)
    if Q(dn_in) < plo:
        dn_in = math.nextafter(dn_in, math.inf)
    dn_out = math.nextafter(dn_in, -math.inf)
    return dn_out, dn_in, up_in, up_out


def expected(fhat: float, lo: Q, hi: Q) -> str:
    plo, phi, fbelow, fabove = L.premise_window(lo, hi)
    F = Q(fhat)
    if plo <= F <= phi:
        return "PASS"
    if F < fbelow or F > fabove:
        return "FAIL"
    return "INCONCLUSIVE"


def show(label, fhat, lo, hi):
    got, exp = L.premise(fhat, lo, hi), expected(fhat, lo, hi)
    f = (lo + hi) / 2
    rel = float(abs(Q(fhat) - f) / abs(f) / Q(1, 2**52))
    ulp = float(abs(Q(fhat) - f) / Q(math.ulp(float(f))))
    print(f"  {label:58s} err {ulp:7.4f} ulp(f) = {rel:6.4f} x 2^-52 rel"
          f"  expected {exp:12s} checker {got:12s} {'ok' if got == exp else 'MISMATCH'}")
    return got == exp


def main():
    ok = True
    print("=== Threshold in ulps depends on the position in the binade ===")
    print("  For normal f in [2^e, 2^(e+1)): ulp = 2^(e-52) and the allowance is")
    print("  UF f + nu = 2^-51 f + 2^-1073, i.e. 2 f / 2^e ulps (+ a subnormal term):")
    print("  between 2 ulps (f = 2^e) and 4 ulps (f -> 2^(e+1)). 3 ulps passes iff f > 1.5 x 2^e.")
    for f in (1.0, 1.25, 1.5, 1.75, math.nextafter(2.0, 0), 0.8414709848078965):
        e = math.frexp(f)[1] - 1
        allow = (L.UF * Q(f) + L.NU) / Q(math.ulp(f))
        print(f"    f = {f!r:22} binade 2^{e}: allowance {float(allow):.6f} ulps")

    print("\n=== Review 01 F2 reproduced: sin(1) + 3 ulps is inside A-FP-2 ===")
    lo, hi = L.sin_enc(1.0)
    ok &= show("sin(1) + 3 ulps (positive control: must PASS)", ulps_from(math.sin(1.0), 3), lo, hi)

    print("\n=== Constructed negative controls: definite violations (must FAIL) ===")
    # an argument whose sine lies just above a power of two: allowance ~ 2 ulps
    x = math.asin(0.5) + 1e-12  # sin x ~ 0.5000000000008660: bottom of the [0.5, 1) binade
    lo, hi = L.sin_enc(x)
    ok &= show("sin(x), sin x = 0.5(1 + 1.7e-12): + 3 ulps (must FAIL)", ulps_from(math.sin(x), 3), lo, hi)
    ok &= show("sin(1) + 5 ulps (allowance 3.37 ulps: must FAIL)", ulps_from(math.sin(1.0), 5),
               *L.sin_enc(1.0))
    lo, hi = L.exp_enc(-0.6931471805599453 + 1e-9)  # exp just above 0.5
    ok &= show("exp(-ln2 + 1e-9) + 3 ulps (bottom of binade: must FAIL)",
               ulps_from(math.exp(-0.6931471805599453 + 1e-9), 3), lo, hi)

    print("\n=== Boundary tests: last double inside / first double outside (closed form) ===")
    for label, (lo, hi), f0 in (("sin(1)", L.sin_enc(1.0), math.sin(1.0)),
                                ("cos(0.3)", L.cos_enc(0.3), math.cos(0.3)),
                                ("atan2(1e5, 6.4e6)", L.atan2_enc(1e5, 6.4e6), math.atan2(1e5, 6.4e6)),
                                ("exp(-0.9)", L.exp_enc(-0.9), math.exp(-0.9))):
        dn_out, dn_in, up_in, up_out = last_inside_and_first_outside(lo, hi)
        ok &= show(f"{label}: largest passing double", up_in, lo, hi)
        ok &= show(f"{label}: next double up", up_out, lo, hi)
        ok &= show(f"{label}: smallest passing double", dn_in, lo, hi)
        ok &= show(f"{label}: next double down", dn_out, lo, hi)

    print("\n=== Subnormal regime: the absolute term nu dominates ===")
    xs = -744.0  # exp(-744) ~ 1.03e-323 = about 2 x 2^-1074
    lo, hi = L.exp_enc(xs)
    f0 = math.exp(xs)
    print(f"  exp(-744) = {f0!r} = {Q(f0) / Q(2) ** -1074} x 2^-1074;"
          f" exact in [{float(lo):.6e}, {float(hi):.6e}]")
    for k in (-2, -1, 1, 2, 3):
        ok &= show(f"exp(-744) {k:+d} subnormal ulps", ulps_from(f0, k), lo, hi)

    print("\n=== Oracle uncertainty: an enclosure straddling the window edge ===")
    lo, hi = L.sin_enc(1.0)
    dn_out, dn_in, up_in, up_out = last_inside_and_first_outside(lo, hi)
    # widen the enclosure artificially so that up_in sits between the PASS and FAIL bounds
    wide_lo = lo - Q(math.ulp(1.0)) * 2
    ok &= show("largest passing double vs a widened enclosure (must be INCONCLUSIVE)", up_in, wide_lo, hi)
    print("  INCONCLUSIVE is treated as PREMISE-NOT-OBSERVED (fail closed), never as PASS.")

    print("\n=== Exact references (identities): zero and one ===")
    for label, fhat, lo, hi, exp_ in (("sin(+0) -> +0.0", 0.0, Q(0), Q(0), "PASS"),
                                      ("sin(+0) -> 2^-1074", 5e-324, Q(0), Q(0), "PASS"),
                                      ("sin(+0) -> 3 x 2^-1074", 1.5e-323, Q(0), Q(0), "FAIL"),
                                      ("exp(0) -> 1.0", 1.0, Q(1), Q(1), "PASS"),
                                      ("exp(0) -> 1 + 2^-52", 1.0 + 2**-52, Q(1), Q(1), "PASS"),
                                      ("exp(0) -> 1 + 3 x 2^-52", 1.0 + 3 * 2**-52, Q(1), Q(1), "FAIL")):
        got = L.premise(fhat, lo, hi)
        print(f"  {label:30s} expected {exp_:5s} checker {got:5s} {'ok' if got == exp_ else 'MISMATCH'}")
        ok &= got == exp_
    print(f"\nALL CONTROLS AGREE WITH THE CLOSED-FORM INEQUALITY: {ok}")


main()
