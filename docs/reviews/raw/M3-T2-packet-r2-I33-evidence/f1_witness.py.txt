Reproduces Review 01 M3T2P2-R1-F1 (a component whose observed trace is complete and

Usage: python -I f1_witness.py <t0_dir>
"""

E1 = tt.rounding_allowances()
ALPHA, BETA = E1["alpha"], E1["beta_m"]
SRC = inspect.getsource(m.geodetic_iterative)


def mutant(old: str, new: str, **env_extra):
    """A copy of geodetic_iterative with one source substitution, in its own namespace."""
    assert SRC.count(old) == 1, old
    env = dict(m.__dict__)
    env.update(env_extra)
    exec(SRC.replace(old, new), env)
    return env["geodetic_iterative"], env


class Tracer:
    """Packet r2's DL-3 design: a module-attribute proxy on the component's `math`."""

    def __init__(self, base=math):
        self.base, self.calls = base, []

    def __getattr__(self, name):
        f = getattr(self.base, name)
        if name not in ("sin", "cos", "atan2", "hypot", "sqrt"):
            return f

        def wrapped(*a):
            out = f(*a)
            self.calls.append((name, a, out))
            return out
        return wrapped


def r2_count_check(fn, env, r):
    """Packet r2 §6.4 rule 4: the traced call list equals the list implied by the
    iteration count. Returns (passes, traced_calls, output)."""
    tr = Tracer()
    old = env["math"]
    env["math"] = tr
    try:
        out = fn(r)
    finally:
        env["math"] = old
    names = [c[0] for c in tr.calls]
    it = names.count("atan2") - 1
    return names == W.expected_geodetic_calls(it), tr.calls, out


class Shifted:
    """A math stand-in whose `name` results are moved by `rel` relative (or by `ulps`)."""

    def __init__(self, name, rel=None, ulps=None, at=None):
        self.name, self.rel, self.ulps, self.at, self.n = name, rel, ulps, at, 0

    def __getattr__(self, attr):
        f = getattr(math, attr)
        if attr != self.name:
            return f

        def g(*a):
            out = f(*a)
            self.n += 1
            if self.at is not None and self.n - 1 != self.at:
                return out
            if self.rel is not None:
                return out * (1.0 + self.rel)
            for _ in range(self.ulps):
                out = math.nextafter(out, math.inf)
            return out
        return g


def point():
    return np.array([m.A_WGS84 + 120000.0, 0.0, 0.0])


def oblique():
    return m.geodetic_to_cartesian(0.6, 0.4, 400000.0)


def run(label, comp, r, base=math):
    out, info = W.witness_geodetic(comp, r, ALPHA, BETA, base=base)
    print(f"  {label:66s} -> {out}")
    return out


def main():
    print("=== F1 reproduction (Review 01): hidden cosine behind a complete trace ===")
    hid = lambda x: math.cos(x) + 1e-5  # noqa: E731
    fa, env_a = mutant("s, c = math.sin(lat), math.cos(lat)",
                       "s, c = math.sin(lat), (math.cos(lat), hidden_cos(lat))[1]", hidden_cos=hid)
    r = point()
    ok_good, calls_good, out_good = r2_count_check(m.geodetic_iterative, m.__dict__, r)
    ok_bad, calls_bad, out_bad = r2_count_check(fa, env_a, r)
    print(f"  r2 count check: correct component {ok_good}, mutant {ok_bad};"
          f" identical traces {calls_good == calls_bad}; height difference {out_bad[1] - out_good[1]!r} m")
    print(f"  call multiset {dict(collections.Counter(c[0] for c in calls_bad))}")

    print("\n=== Witnessed evaluation: the I-33 §2 adversarial designs ===")
    for r, name in ((point(), "equator 120 km"), (oblique(), "lat 0.6, 400 km")):
        print(f" point: {name}")
        run("correct component (T0 spike)", m.geodetic_iterative, r)
        run("(1) correct trace, hidden incorrect cosine (+1e-5)", fa, r)
        f1b, _ = mutant("s, c = math.sin(lat), math.cos(lat)",
                        "s, c = math.sin(lat), (math.cos(lat), math.nextafter(math.cos(lat), 2.0))[1]")
        run("(1b) hidden cosine 1 ulp high (within A-FP-2)", f1b, r)
        f2, _ = mutant("s, c = math.sin(lat), math.cos(lat)", "_d, c = math.sin(lat), math.cos(lat)")
        run("(2) final sin traced but its value discarded (stale s used)", f2, r)
        f2b, _ = mutant("    p = math.hypot(x, y)\n", "    p = math.hypot(x, y)\n    p = abs(x) + abs(y) * 0.0\n")
        run("(2b) hypot traced, value replaced after tracing", f2b, r)
        cos_bound = math.cos
        f3, _ = mutant("s, c = math.sin(lat), math.cos(lat)",
                       "s, c = math.sin(lat), (math.cos(lat), _cos(lat))[1]", _cos=cos_bound)
        run("(3a) unobserved call supplies the value: same function bound at import", f3, r)
        f3b, _ = mutant("s, c = math.sin(lat), math.cos(lat)",
                        "s, c = math.sin(lat), (math.cos(lat), float(_np.cos(lat)))[1]", _np=np)
        run("(3b) unobserved call supplies the value: numpy.cos", f3b, r)
        f3c, _ = mutant("s, c = math.sin(lat), math.cos(lat)",
                        "s, c = math.sin(lat), (math.cos(lat), _cos(lat) * (1 + 2.0**-40))[1]",
                        _cos=cos_bound)
        run("(3c) unobserved perturbed call (2^-40 relative) supplies the value", f3c, r)
        f4, env4 = mutant("s, c = math.sin(lat), math.cos(lat)", "s, c = math.sin(lat), _cos(lat)",
                          _cos=cos_bound)
        ok4 = r2_count_check(f4, env4, r)[0]
        out4 = run("(4) component omits one traced invocation (cos bound at import)", f4, r)
        print(f"      (r2 count check on this mutant: {'passes' if ok4 else 'rejects'}; the witness"
              f" decision does not depend on tracing the component)")

    print("\n=== (4') harness omission: a twin that does not route one call through lib ===")
    src_twin = inspect.getsource(W.twin_geodetic)
    old = "c = lib.cos(lat); k(c)"
    assert src_twin.count(old) == 1
    env = dict(W.__dict__)
    exec(src_twin.replace(old, "c = math.cos(lat); k(c)"), env)
    bad_twin = env["twin_geodetic"]
    r = oblique()
    lib = W.CheckedLib()
    lat, h, it, _ = bad_twin(*r, lib)
    names = [c[0] for c in lib.calls]
    print(f"  omission twin: {len(names)} recorded calls, expected {len(W.expected_geodetic_calls(it))}"
          f" -> completeness assertion {'KILLS' if names != W.expected_geodetic_calls(it) else 'MISSES'} the mutant")

    print("\n=== Fault injection at every call index of the correct twin (completeness proof by test) ===")
    for r, name in ((point(), "equator 120 km"), (oblique(), "lat 0.6, 400 km")):
        lib = W.CheckedLib()
        _, _, it, _ = W.twin_geodetic(*r, lib)
        n = len(lib.calls)
        detected = 0
        for idx in range(n):
            class Inject:  # a definite premise violation at call idx only: relative 2^-30
                def __init__(self):
                    self.k = 0

                def __getattr__(self, a):
                    f = getattr(math, a)

                    def g(*args):
                        out = f(*args)
                        self.k += 1
                        if self.k - 1 != idx:
                            return out
                        return out * (1.0 + 2.0**-30) if out != 0 else 2.0**-1000
                    return g
            inj = Inject()
            # component = the twin itself on the same faulty library: conformance holds by
            # construction, so only the premise check can reject
            comp = lambda rr: W.twin_geodetic(*rr, inj2)[:2]  # noqa: E731
            inj2 = Inject()
            out, _ = W.witness_geodetic(comp, r, ALPHA, BETA, base=inj)
            detected += out.startswith("PREMISE-NOT-OBSERVED")
        print(f"  {name}: {it} iterations, {n} calls; violation at each index detected: {detected}/{n}")

    print("\n=== F5: a faulty math shared by component and twin ===")
    faulty = Shifted("cos", rel=1e-5 / 0.9)  # about +1e-5 on cos near 0.9
    r = oblique()
    envc = dict(m.__dict__)
    envc["math"] = faulty
    exec(SRC, envc)
    lib = W.CheckedLib(Shifted("cos", rel=1e-5 / 0.9))
    lat_t, h_t, _, _ = W.twin_geodetic(*r, lib)
    lat_c, h_c = envc["geodetic_iterative"](r)
    print(f"  bitwise conformance component vs twin: {W.bits(lat_c, h_c) == W.bits(lat_t, h_t)};"
          f" height error {float(h_c - Q(m.geodetic_iterative(r)[1])):.6g} m")
    run("witness decision (shared faulty cos)", envc["geodetic_iterative"], r,
        base=Shifted("cos", rel=1e-5 / 0.9))

    print("\n=== Native hypot: contract observed, internals not ===")
    r = oblique()
    for label, sh in (("hypot +1 ulp (inside A-FP-2)", Shifted("hypot", ulps=1)),
                      ("hypot x (1 + 2^-40) (outside A-FP-2)", Shifted("hypot", rel=2.0**-40))):
        envh = dict(m.__dict__)
        envh["math"] = sh
        exec(SRC, envh)
        run(label + ", shared", envh["geodetic_iterative"], r,
            base=Shifted(sh.name, rel=sh.rel, ulps=sh.ulps))

    print("\n=== Instrumentation neutrality: twin through CheckedLib vs twin on raw math ===")
    pts = [point(), oblique()] + [m.geodetic_to_cartesian(la, lo, h) for la, lo, h in
                                  ((-1.2, 2.0, 1e3), (1.5, -0.5, 2e6), (1e-9, 0.1, 4e5))]
    same = sum(W.bits(*W.twin_geodetic(*p, W.CheckedLib())[:2]) == W.bits(*W.twin_geodetic(*p, math)[:2])
               for p in pts)
    conf = sum(W.bits(*W.twin_geodetic(*p, math)[:2]) == W.bits(*m.geodetic_iterative(p)) for p in pts)
    print(f"  identical bits: {same}/{len(pts)}; twin (text transcription) vs T0 spike bitwise: {conf}/{len(pts)}")


main()
