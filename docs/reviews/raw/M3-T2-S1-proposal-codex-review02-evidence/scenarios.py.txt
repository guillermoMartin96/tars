import subprocess,os,sys
from pathlib import Path
p=Path('review_scratch/reviewer')
for tag,src in [('base','src'),('proto',str(p/'proto/src'))]:
 for s in sorted(Path('scenarios').glob('*.json')):
  r=subprocess.run([sys.executable,'-B','toolbox/scripts/run_mission.py','--scenario',str(s),'--out',str(p/'missions'/tag/s.stem)],env={**os.environ,'PYTHONPATH':src,'PYTHONDONTWRITEBYTECODE':'1'},capture_output=True,text=True)
  print(tag,s.name,r.returncode,r.stdout,r.stderr,flush=True)
for s in sorted(Path('scenarios').glob('*.json')):
