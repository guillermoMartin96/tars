"""I-28 self-validation (Review 02 corrections). Run from the repository root
after `git add -A`. Compares the staged tree with BASE (45f80f8, Review 02
record) and with A22 (a22297d, the I-27 base). Prints PASS/FAIL per check; the
final line is the overall result. Labels are right-stripped (R2-F2 lesson)."""
import glob, re, subprocess, sys
from pathlib import Path

BASE, A22 = "45f80f8523d1ed0033cf9f773d5072e719d4d442", "a22297d1ec719a58aeb7209f3d34256fa7b0756a"
repo = Path(".")
fails = []


def git(*a, check=True):
    r = subprocess.run(["git", *a], capture_output=True, text=True)
    if check and r.returncode:
        raise SystemExit(r.stderr)
    return r


def bad(msg):
    fails.append(msg)
    print(f"    FAIL: {msg}")


def show(rev, f):
    return git("show", f"{rev}:{f}").stdout


def showb(rev, f):
    return subprocess.run(["git", "show", f"{rev}:{f}"], capture_output=True, check=True).stdout


# [1] R2-F1: the two cells carry the marker; removing it restores the original line exactly
T0 = "docs/milestones/M3-T0-decision-matrix.md"
old, new = show(BASE, T0).split("\n"), (repo / T0).read_text().split("\n")
mk = re.compile(r" \*\(Superseded in part by I-28 \(M3T2OD-R2-F1\).*?\"I-27 correction\"\.\)\*")
need = ["`reentry` is **DECIDED**", "`non_finite` → `reentry` → `impact` → `escape` is **DECIDED**",
        "does not decide them again", "only within the approved contract"]
for ln in (14, 27):
    restored = mk.sub("", new[ln - 1], count=1)
    ok = restored == old[ln - 1] and all(n in new[ln - 1] for n in need)
    print(f"[1] {'ok  ' if ok else 'FAIL'} {T0}:{ln} marker present, original text preserved, required statements present")
    if not ok:
        bad(f"R2-F1 cell {ln}")
others = [i for i in range(len(old)) if i not in (13, 26) and old[i] != new[i]]
print(f"[1] {'ok  ' if not others and len(old) == len(new) else 'FAIL'} no other line of {T0} changed")
if others or len(old) != len(new):
    bad("other T0 matrix lines changed")

# [2] full-document search, every line (not only added lines), single-line and +/-4-line windows
files = sorted(glob.glob("docs/decisions/requests/DR-002[4-9]*.md") + glob.glob("docs/decisions/0009-*.md")
               + glob.glob("docs/milestones/M3-*.md") + ["README.md"])
openish = re.compile(r"propos|candidate|pending|confirm|\bopen\b|undecided|until|\bat T3\b|design at T3|provisional|deferred|awaiting", re.I)
subject = re.compile(r"reentry|outcome name|precedence|\bnames\b", re.I)
marker = re.compile(r"I-2[78]\b|superseded|\*\*DECIDED\*\*|Proposal accepted", re.I)
hits = []
for f in files:
    L = (repo / f).read_text().split("\n")
    near = {j for i, l in enumerate(L) if "reentry" in l for j in range(max(0, i - 4), min(len(L), i + 5))}
    for i in sorted(near):
        if openish.search(L[i]) and subject.search(L[i]) and not marker.search(L[i]):
            hits.append((f, i + 1))
# Every unmarked hit is classified by hand (I-28 record, "Full search"). Unknown hits fail.
classified = {
    ("docs/decisions/requests/DR-0024-m3-physical-models-and-constants.md", 222): "historical Q8 question text (pre-T0)",
    ("docs/decisions/requests/DR-0024-m3-physical-models-and-constants.md", 319): "historical Q8 question text (pre-T0)",
    ("docs/decisions/requests/DR-0027-m3-atmosphere-validity-boundary-and-table-residuals.md", 93): "historical A-5 question text; transitioned in the I-26 note A-5 row",
    ("docs/decisions/requests/DR-0027-m3-atmosphere-validity-boundary-and-table-residuals.md", 96): "superseded by the row-level transition at line 146 (I-27 marker names this row)",
    ("docs/decisions/requests/DR-0029-P1-P9-owner-briefing.md", 12): "pre-decision question text; answered at line 122 (Proposal accepted)",
    ("docs/decisions/requests/DR-0029-P1-P9-owner-briefing.md", 89): "pre-decision question text; answered at line 122 (Proposal accepted)",
    ("docs/decisions/requests/DR-0029-P1-P9-owner-decision-packet.md", 26): "pre-decision packet assessment, preserved as reviewed; I-26 section points to DR-0029",
    ("docs/decisions/requests/DR-0029-P1-P9-owner-decision-packet.md", 173): "pre-decision alternatives analysis",
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", 221): "NEW CONTRADICTION M3T2OD-I28-C1 (not corrected; STOP)",
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", 840): "review-history log entry describing R1-F1",
    ("docs/milestones/M3-T2-authorization-packet-r1.md", 296): "review-history log entry describing R1-F1",
    ("docs/milestones/M3-open-issues-register.md", 62): "review-history log entry describing R1-F1",
    ("docs/milestones/M3-proposal.md", 77): "false positive ('proposal ... precedence' in G0 text)",
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", 21): "states DECIDED (I-26) for the precedence",
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", 77): "false positive (stress test, 'not a proposed vehicle')",
    ("docs/milestones/M3-T0-decision-matrix.md", 15): "historical options/recommendation columns of P3; no status claim; I-26/I-27 notes transition P3",
    ("docs/milestones/M3-decision-matrix.md", 337): "I-27 note stating DECIDED ('reconfirm' matched)",
}
for h in hits:
    c = classified.get(h)
    print(f"[2] {'ok  ' if c else 'FAIL'} {h[0]}:{h[1]} -> {c or 'UNCLASSIFIED'}")
    if not c:
        bad(f"unclassified unmarked hit {h}")
for h in classified:
    if h not in hits:
        print(f"[2] note: classified line no longer a hit: {h[0]}:{h[1]}")
print(f"[2] files scanned={len(files)} unmarked hits={len(hits)}; new contradictions={sum('NEW' in classified.get(h, '') for h in hits)}")

# [3] section 4 audit: every decision document other than the T0 matrix is byte-identical to BASE and to A22
changed = git("diff", "--cached", "--name-only", BASE).stdout.split()
dec = [f for f in files if f != T0]
for f in dec:
    if f in changed:
        bad(f"decision document changed: {f}")
print(f"[3] {'ok  ' if not any(f in changed for f in dec) else 'FAIL'} {len(dec)} decision documents byte-identical to {BASE[:7]}")
a22diff = [f for f in dec if show(A22, f) != (repo / f).read_text()]
print(f"[3] info: decision documents differing from {A22[:7]} (I-27 corrections only): {len(a22diff)}")
anchors = [
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", r"^\| P[1-9] \| \*\*(DECIDED|NOT FULLY DECIDED)", "Resolution status rows"),
    ("docs/decisions/requests/DR-0028-P6b-owner-ruling-packet.md", r"Nothing here is decided", "P6(b) ruling not decided (packet status)"),
    ("docs/decisions/requests/DR-0029-m3-t0-proposals.md", r"RM-3b is an unmeasured candidate for evaluation during T4", "RM-3b unmeasured (I-27)"),
    ("docs/milestones/M3-T2-authorization-packet-r1.md", r"report-only", "packet r1 report-only rows"),
]
for f, pat, what in anchors:
    m = [l.rstrip() for l in (repo / f).read_text().split("\n") if re.search(pat, l)]
    print(f"[3] {'ok  ' if m else 'FAIL'} {what}: {len(m)} line(s) in {f}")
    if not m:
        bad(f"anchor missing: {what}")
    if what == "Resolution status rows":
        for l in m:
            print(f"      {l[:70].rstrip()}")

# [4] historical review evidence and records not rewritten
hist = git("ls-tree", "-r", "--name-only", BASE, "docs/reviews").stdout.split()
appended = "docs/reviews/M3-T2-owner-decisions-review-01-corrections.md"
rew = []
for f in hist:
    o, n = showb(BASE, f), (repo / f).read_bytes()
    if f == appended:
        if not n.startswith(o):
            rew.append(f)
    elif o != n:
        rew.append(f)
print(f"[4] {'ok  ' if not rew else 'FAIL'} {len(hist)} files under docs/reviews at {BASE[:7]}: unchanged (one append-only: {appended})")
for f in rew:
    bad(f"historical review file rewritten: {f}")

# [5] scope: documentation only, no protected path
prot = ("src/", "tests/", "proof/", "toolbox/", "scenarios/", ".github/", "uv.lock", "pyproject.toml")
outside = [f for f in changed if not f.startswith("docs/")]
protected = [f for f in changed if f.startswith(prot)]
print(f"[5] {'ok  ' if not outside and not protected else 'FAIL'} {len(changed)} changed files, all under docs/; protected paths changed: {len(protected)}")
for f in changed:
    print(f"      {f}")
if outside or protected:
    bad("scope")

# [6] links in changed Markdown resolve
for f in [c for c in changed if c.endswith(".md")]:
    for t in re.findall(r"\]\(([^)#]+)", (repo / f).read_text()):
        if not t.startswith("http") and not (Path(f).parent / t).exists():
            bad(f"broken link {f} -> {t}")
print(f"[6] {'ok  ' if not any('broken link' in x for x in fails) else 'FAIL'} relative links in changed Markdown resolve")

# [7] fresh whitespace check of the I-28 changes; exit status reported as returned
r = git("diff", "--cached", "--check", BASE, check=False)
print(f"[7] git diff --cached --check {BASE[:7]}: exit {r.returncode}")
for l in r.stdout.rstrip().split("\n") if r.stdout.strip() else []:
    print(f"      {l.rstrip()[:110].rstrip()}")
if r.returncode:
    bad("whitespace check of I-28 changes")

print(f"RESULT: {'PASS' if not fails else 'FAIL'} ({len(fails)} failure(s))")
sys.exit(1 if fails else 0)
