def bits(x):
    return struct.pack('>d', x)

tiny = math.ulp(0.0)
values = [0., -0., tiny, -tiny, 2**-1022, -(2**-1022), 1., -1., 1e200, -1e200,
          sys.float_info.max, -sys.float_info.max, math.inf, -math.inf,
          struct.unpack('>d', bytes.fromhex('7ff8000000000123'))[0]]
even_fail = swap_fail = 0
for x in values:
    for y in values:
        baseline = bits(math.hypot(x,y))
        swap_fail += baseline != bits(math.hypot(y,x))
        even_fail += any(baseline != bits(math.hypot(sx*x,sy*y)) for sx in [-1,1] for sy in [-1,1])
print('adversarial hypot pairs', len(values)**2, 'even failures', even_fail, 'swap failures', swap_fail)
