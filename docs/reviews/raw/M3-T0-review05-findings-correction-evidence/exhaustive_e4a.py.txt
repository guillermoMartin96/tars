"""Exhaustive check of E4a on small subnormal grids (implementer evidence, not a test)."""
import sys
from fractions import Fraction
sys.path.insert(0, "docs/science/experiments/m3-t0")
import t0_termination as tt
T = 2.0**-1074
PHI = Fraction(tt.FP["phi_B"])
shapes = [(2.2, 15.0), (1.0, 1.0), (0.5, 3.0), (2.2, 2 * T), (2.2, 7 * T), (1e-160, 1e-160), (1e-300, 100.0), (3.0, 1e-310)]
worst = {}
n = viol = 0
for cd, area in shapes:
    p = cd * area
    w = Fraction(0)
    for k in range(1, 20001):
        mass = k * T
        if p > 0:  # C1
            b = mass / p
            if 0 < b < float("inf"):
                exact = Fraction(cd) * Fraction(area) / Fraction(mass)
                r = tt.inverse_ballistic_bound(b, cd, area, PHI)
                n += 1; viol += exact > r; w = max(w, exact / r)
        b = k * T  # C2
        m2 = (b * cd) * area
        if 0 < m2 < float("inf"):
            exact = Fraction(cd) * Fraction(area) / Fraction(m2)
            r = tt.inverse_ballistic_bound(b, cd, area, PHI)
            n += 1; viol += exact > r; w = max(w, exact / r)
    worst[(cd, area)] = float(w)
print("vehicles", n, "violations", viol)
for k, v in worst.items():
    print(k, "max exact/R", v)
