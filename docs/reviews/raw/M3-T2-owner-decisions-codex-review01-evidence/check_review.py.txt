def packet_density(h):
    if not math.isfinite(h): raise ValueError('nonfinite')
    if h<0: raise m.OutsideDomainError()
    h=h/1000.; b=m.TABLE[m.band_index(h)]
    return b.rho0*math.exp(-(h-b.base_km)/b.scale_km)
def packet_geod(r):
    x,y,z=(float(v) for v in r);p=math.hypot(x,y)
    lat=math.atan2(z,p*(1.-m.E2_WGS84))
    for _ in range(60):
        s=math.sin(lat);n=m.A_WGS84/math.sqrt(1.-m.E2_WGS84*s*s)
        new=math.atan2(z+m.E2_WGS84*n*s,p)
        done=abs(new-lat)<=1e-15;lat=new
        if done: break
    s,c=math.sin(lat),math.cos(lat)
    return lat,p*c+z*s-m.A_WGS84*math.sqrt(1.-m.E2_WGS84*s*s)
