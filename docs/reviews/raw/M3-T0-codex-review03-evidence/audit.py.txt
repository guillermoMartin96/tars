out['historical_reviews_preserved']={p:subprocess.check_output(['git','show','5548d11:'+p])==pathlib.Path(p).read_bytes() for p in ['docs/reviews/raw/M3-T0-codex-review01.md','docs/reviews/raw/M3-T0-codex-review02.md','docs/reviews/raw/M3-T0-codex-review01-prompt.md','docs/reviews/raw/M3-T0-codex-review02-prompt.md']}
