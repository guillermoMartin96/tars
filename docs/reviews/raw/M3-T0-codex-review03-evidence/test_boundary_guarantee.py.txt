import sys
from pathlib import Path
import numpy as np
sys.path.insert(0,str(Path(__file__).resolve().parents[1]/'docs/science/experiments/m3-t0'))
import t0_model as m
import t0_termination as t

def test_declared_envelope_covers_the_model_at_a_downward_jump():
    # True height independently verified to 80 digits by counterexample.py; speed < 11000.
    r=np.array([5420209.728043159,3584286.6758661848,1.8499824933775928])
    v=np.array([6067.454938935459,-9175.29239664793,0.])
    actual=t.drag_only(t.J2_DRAG,r,v,t.MASS)
    bound=t.drag_envelope(120000.,11000.,t.B_V1)
    assert actual <= bound + 1e-12, (actual,bound,m.geodetic_iterative(r)[1])
