files=git('diff','--name-only',base,'HEAD').splitlines()
print('changed_files',len(files),'docs_only',all(x.startswith('docs/') for x in files))
for p in ['docs/milestones/M3-T2-authorization-packet.md','docs/milestones/M3-T2-decision-readiness-report.md']:
 old=subprocess.check_output(['git','show',base+':'+p]); print('prefix',p,Path(p).read_bytes().startswith(old))

 r=m.geodetic_to_cartesian(rng.uniform(-1.57,1.57),rng.uniform(-math.pi,math.pi),rng.uniform(1,2e6))
 x,y,z=map(float,r); h=m.geodetic_iterative(r)[1]; upper=(F(h)+b)/(1-a); bound=2*(a*upper+b)
 for s in [(-y,x,z),(-x,-y,z),(y,-x,z),(x,y,-z)]:
  hs=m.geodetic_iterative(s)[1]; diff=abs(F(h)-F(hs)); bfail+=diff>bound; maxdiff=max(maxdiff,float(diff))
  if s!=(x,y,-z): qfail+=h!=hs
print('samples quarter_fail bounded_fail max_reflection_difference',2000,qfail,bfail,maxdiff)
