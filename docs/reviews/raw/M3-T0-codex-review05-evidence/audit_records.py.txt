import pathlib,json,hashlib,re,subprocess
R=pathlib.Path.cwd();S=R/'review_scratch';D=R/'docs/science/experiments/m3-t0'

inv=[]
for f in sorted(set(files)):
 if not f.is_file():continue
 text=f.read_text(); inv.append({'path':str(f.relative_to(R)),'bytes':f.stat().st_size,'sha256':hashlib.sha256(f.read_bytes()).hexdigest()})

 for name in names:
  p=D/name
  if p.is_file():
   h=hashlib.sha256(p.read_bytes()).hexdigest();checks.append({'file':name,'digest':h,'claimed':hs,'match':h in hs})
