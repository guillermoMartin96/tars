"""Implementer reproduction of Review 07 M3T0-R7-F1 and R7-F2 (no correction applied)."""
import math
import sys
from fractions import Fraction

sys.path.insert(0, sys.argv[1])
import t0_termination as tt  # noqa: E402

TINY = math.ulp(0.0)
print("R7-F1: infinite envelope with the early return (M2 <= 0)")
for h0, b in ((1.0, TINY), (1e3, TINY), (1e7, TINY), (1.0, 0.0), (1e7, 0.0)):
    lim = tt.stage_lower_heights(h0, b)
    need = tt.stage_need(h0, b)
    print(f"  h0={h0:g} b={b!r}: A0={lim['A0']!r} M2={lim['M2']!r} M3={lim['M3']!r} "
          f"M4={lim['M4']!r} need={need!r} certified={need < 0.0}")
print("R7-F2: b_R <= b at b = 0")
b_r = tt.envelope_ballistic_coefficient(0.0, 2.2, 15.0, tt.FP)
print(f"  b_R(0, 2.2, 15) = {b_r} ; b_R <= b: {Fraction(b_r) <= 0}")
for b in (TINY, 2 * TINY, 1e-310, 25.757575757575758):
    print(f"  b={b!r}: b_R <= b: {Fraction(tt.envelope_ballistic_coefficient(b, 2.2, 15.0, tt.FP)) <= Fraction(b)}")
