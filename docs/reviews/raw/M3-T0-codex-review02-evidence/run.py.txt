import subprocess,pathlib,hashlib,json,time
S=pathlib.Path('review_scratch'); D=pathlib.Path('docs/science/experiments/m3-t0'); ledger=[]
def run(args,name):
 t=time.time(); p=subprocess.run(args,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True); (S/(name+'.log')).write_text(p.stdout); ledger.append({'command':args,'exit':p.returncode,'seconds':time.time()-t,'log':name+'.log'}); (S/'commands.json').write_text(json.dumps(ledger,indent=2)); print(name,p.returncode,p.stdout[-400:],flush=True)
before={str(p):hashlib.sha256(p.read_bytes()).hexdigest() for p in D.rglob('*') if p.is_file() and '__pycache__' not in str(p)}; (S/'before.json').write_text(json.dumps(before,indent=2))
for x in ['atmosphere','decay','geodetic','frames','regression','termination']:
 run(['.venv/bin/python',str(D/f't0_{x}.py'),str(D/f't0_{x}_results.json')],x)
run(['.venv/bin/python',str(D/'t0_gmat.py'),'--gmat-console',str((pathlib.Path('..')/'GMAT R2026a/bin/GmatConsole').resolve()),'--out',str(D/'t0_gmat_results.json')],'gmat')
run(['.venv/bin/python','-m','pytest','-q','-p','no:cacheprovider',str(D/'check_review01.py')],'checks')
run(['.venv/bin/python','-m','pytest','-q','-p','no:cacheprovider'],'pytest')
after={k:hashlib.sha256(pathlib.Path(k).read_bytes()).hexdigest() for k in before}; (S/'reproduction.json').write_text(json.dumps({k:[before[k],after[k]] for k in before if before[k]!=after[k]},indent=2)); print('changed:',(S/'reproduction.json').read_text())
