import sys,math,json,pathlib
import numpy as np
from scipy.optimize import minimize_scalar,brentq
D=pathlib.Path('docs/science/experiments/m3-t0'); sys.path.insert(0,str(D.resolve()))
import t0_model as m, t0_termination as tt
out={}; rng=np.random.default_rng(1202)
# Analytic principal curvatures, independently assembled in local north/east basis.
ratios=[]; speedslack=[]; jrat=[]; normalmax=math.asin((m.A_WGS84**2-m.B_WGS84**2)/(m.A_WGS84**2+m.B_WGS84**2))
for i in range(2000):
 lat=rng.uniform(-math.pi/2,math.pi/2);lon=rng.uniform(-math.pi,math.pi);h=rng.uniform(0,2e6)
 r=m.geodetic_to_cartesian(lat,lon,h); n=np.array([math.cos(lat)*math.cos(lon),math.cos(lat)*math.sin(lon),math.sin(lat)])
 north=np.array([-math.sin(lat)*math.cos(lon),-math.sin(lat)*math.sin(lon),math.cos(lat)]);east=np.array([-math.sin(lon),math.cos(lon),0])
 N=m.A_WGS84/math.sqrt(1-m.E2_WGS84*math.sin(lat)**2);M=m.A_WGS84*(1-m.E2_WGS84)/(1-m.E2_WGS84*math.sin(lat)**2)**1.5
 H=np.outer(north,north)/(M+h)+np.outer(east,east)/(N+h)
 v=rng.normal(size=3);v*=math.sqrt(2*m.MU/np.linalg.norm(r))*rng.uniform(0,1)/np.linalg.norm(v)
 acc=m.acceleration(tt.J2_DRAG,0,r,v,850)
 exact=n@acc+v@H@v; U=tt.hddot_upper(r,v,acc);ratios.append(U-exact)
 speedslack.append(tt.vrel_bound(h)-np.linalg.norm(v-m.air_velocity(r,m.OMEGA_E)))
 jrat.append(np.linalg.norm(m.j2_accel(r,m.J2_FROM_C20,m.MU,m.A_WGS84))/(m.MU/np.linalg.norm(r)**2)/(3*m.J2_FROM_C20*(m.A_WGS84/np.linalg.norm(r))**2))
out.update(analytic_normal_angle=normalmax,code_normal_angle=tt.DELTA_MAX,min_U_minus_exact_hddot=min(ratios),min_speed_slack=min(speedslack),max_J2_bound_ratio=max(jrat))
# Independently refine the U maximum that the sweep labels a supremum.
hp,ha,inc=250e3,800e3,97.;sma,ecc=tt.orbit_elements(hp,ha);period=2*math.pi*math.sqrt(sma**3/m.MU)
r,v=m.perigee_state(hp,ha,inc); sol=tt.dense(tt.J2_DRAG,np.r_[r,v,850],2*period+20)
def U(t):
 y=sol(t); return tt.hddot_upper(y[:3],y[3:6],m.acceleration(tt.J2_DRAG,t,y[:3],y[3:6],y[6]))

out['sampled_sup_counterexample']={'sample_t':ts[k],'sample_U':us[k],'refined_t':opt.x,'refined_U':-opt.fun,'difference':-opt.fun-us[k]}
