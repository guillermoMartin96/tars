   elif k not in b:diffs.append(['REMOVED',path+'/'+k])
   else:compare(a[k],b[k],path+'/'+k)
 elif a!=b:diffs.append(['CHANGED',path,a,b])

# Validate all literal full sha256 digest assertions in VAL-0019 against local files.
import re
text=(root/'docs/science/validation.md').read_text();text=text[text.index('## VAL-0019'):]
checked=[]
for line in text.splitlines():
 matches=re.findall(r'`([^`]+)`',line)
 for x in matches:
  if re.fullmatch('[0-9a-f]{64}',x):
   candidates=[s for s in matches if (root/s).is_file()]
   for c in candidates:
    got=hashlib.sha256((root/c).read_bytes()).hexdigest();checked.append((c,x,got,got==x))
print('VAL digest matches',json.dumps(checked,indent=2))
