import subprocess,json,time,pathlib,sys
p=pathlib.Path('review_scratch'); ledger=p/'commands.jsonl'
cmd=sys.argv[2:]; start=time.time()
r=subprocess.run(cmd,stdout=subprocess.PIPE,stderr=subprocess.STDOUT,text=True)
(p/sys.argv[1]).write_text(r.stdout)
with ledger.open('a') as f:f.write(json.dumps(dict(command=cmd,exit=r.returncode,seconds=time.time()-start,log=sys.argv[1]))+'\n')
print(sys.argv[1],r.returncode,r.stdout[-1200:])
