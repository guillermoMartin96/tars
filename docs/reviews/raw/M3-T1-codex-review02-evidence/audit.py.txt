import subprocess,json,pathlib,hashlib,importlib.util
P=pathlib.Path
out={}
def git(*a):return subprocess.check_output(['git',*a],text=True)
out['head']=git('rev-parse','HEAD').strip();out['status']=git('status','--porcelain');out['protected_diff']=git('diff','60648b9..HEAD','--','proof','toolbox','scenarios','.github','uv.lock','pyproject.toml','src/tars/validation');out['src_diff']=git('diff','60648b9..HEAD','--','src');out['later']=git('show','--stat','1cf0ce9');out['golden_sha256']=hashlib.sha256(P('proof/references/m1_final_state.json').read_bytes()).hexdigest()
for name in ['correction','info']:
 a=P('review_scratch/'+name+'.json'); b=P('docs/science/experiments/m3-t1/'+('t1_review01_correction_check_results.json' if name=='correction' else 't1_j2_informational_results.json'))
 out[name+'_byte_equal']=a.exists() and a.read_bytes()==b.read_bytes()
out['ci']=[]
for p in sorted(P('review_scratch/ci').glob('*.json')):

# Historical prefix check is stronger than line-based added-only inspection.
out['append_only']={}
