for phi in [.1,.3,.6,.8,1.]:
 r=m.geodetic_to_cartesian(phi,0.,120000.)
 lat,h=m.geodetic_iterative(r)
 p=math.hypot(r[0],r[1]); z=float(r[2]);s=math.sin(lat);c=math.cos(lat)
 # The only altered operation is a permitted two-ulp upward cos value at positive lat.
 asymmetric=p*math.nextafter(math.nextafter(c,math.inf),math.inf)+z*s-m.A_WGS84*math.sqrt(1-m.E2_WGS84*s*s)
 print('phi',phi,'height_original',h,'height_allowed_asymmetric_cos',asymmetric,'difference',asymmetric-h)
