"""R9-F2 impact: re-classify the seed-20261022 probe states (same generator as adversarial_r8)
on the E2-defined ellipsoid B^2 = A^2 (1 - E2). Usage: python -I r9f2_probe_reclassify.py <m3-t0 dir>"""
import math
import random
import sys
from fractions import Fraction

import numpy as np

sys.path.insert(0, sys.argv[1])
import t0_model as m  # noqa: E402

rng = random.Random(20261022)
A = Fraction(m.A_WGS84)
BF2 = (A * (1 - Fraction(m.F_WGS84))) ** 2
BE2 = A * A * (1 - Fraction(m.E2_WGS84))
n = ext_f = ext_e = only_f = only_e = 0
for _ in range(1500):
    lat = math.asin(rng.uniform(-1.0, 1.0)); lon = rng.uniform(-math.pi, math.pi)
    base = m.geodetic_to_cartesian(lat, lon, 0.0)
    sign = np.sign(base)
    for k in range(0, 41, rng.choice((1, 2, 3))):
        r = base.copy()
        for i in range(3):
            for _ in range(k):
                r[i] = math.nextafter(r[i], math.inf * sign[i]) if sign[i] else r[i]
        x, y, z = (Fraction(float(c)) for c in r)
        q = (x * x + y * y) / (A * A)
        f, e = q + z * z / BF2 > 1, q + z * z / BE2 > 1
        n += 1; ext_f += f; ext_e += e; only_f += f and not e; only_e += e and not f
print(dict(states=n, exterior_F=ext_f, exterior_E2=ext_e, exterior_F_only=only_f, exterior_E2_only=only_e))
